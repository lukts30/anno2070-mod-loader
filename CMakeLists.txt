cmake_minimum_required(VERSION 3.24)
project(anno2070_modloader VERSION 0.1.0 LANGUAGES C CXX)
include(FetchContent)
FetchContent_Declare(xmlcore URL https://github.com/jakobharder/anno-mod-loader/archive/5a4464dc2da222844f439ac0463174d2da97b087.tar.gz)
FetchContent_Declare(pugixml URL https://github.com/zeux/pugixml/archive/ee86beb30e4973f5feffe3ce63bfa4fbadf72f38.tar.gz)
FetchContent_Declare(spdlog URL https://github.com/gabime/spdlog/archive/76fb40d95455f249bd70824ecfcae7a8f0930fa3.tar.gz)
FetchContent_Declare(minhook URL https://github.com/TsudaKageyu/minhook/archive/c3fcafdc10146beb5919319d0683e44e3c30d537.tar.gz)
FetchContent_MakeAvailable(xmlcore pugixml spdlog minhook)
file(GLOB core_sources CONFIGURE_DEPENDS "${xmlcore_SOURCE_DIR}/libs/xml-operations/src/*.cc")
add_library(version SHARED src/loader.cpp src/proxy.cpp src/version.def ${core_sources})
target_compile_features(version PRIVATE cxx_std_17)
target_include_directories(version PRIVATE "${xmlcore_SOURCE_DIR}/libs/xml-operations/include" "${spdlog_SOURCE_DIR}/include")
target_link_libraries(version PRIVATE pugixml minhook user32)
set_target_properties(version PROPERTIES PREFIX "")
if(MINGW)
  target_compile_definitions(version PRIVATE strnicmp=_strnicmp)
  target_compile_options(version PRIVATE "SHELL:-include cstring" "SHELL:-include algorithm" "SHELL:-include functional" "SHELL:-include vector")
  target_link_options(version PRIVATE -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp)
endif()
install(TARGETS version RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES "${xmlcore_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME xmlcore.txt)
install(FILES "${pugixml_SOURCE_DIR}/LICENSE.md" DESTINATION licenses RENAME pugixml.txt)
install(FILES "${spdlog_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME spdlog.txt)
install(FILES "${minhook_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME minhook.txt)
